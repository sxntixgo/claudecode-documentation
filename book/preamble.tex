% book/preamble.tex — packages and global settings
%
% Visual style (typography, code blocks, admonition boxes, list markers,
% section headings) is provided by `gimenezocano.sty', which must be
% loaded before everything else so its font, color, and tcolorbox setup
% is in place when later packages run.
%
% Pass [print] for a B&W print-ready build:
%   \usepackage[print]{gimenezocano}
\usepackage{gimenezocano}

% Unicode input — required for em-dashes (—), curly quotes, etc., that
% appear in source text and inside lstlisting blocks. The book uses
% pdflatex; without this, characters outside Latin-1 cause compile errors.
\usepackage[utf8]{inputenc}

% The `listings' package (loaded by gimenezocano.sty) does not handle UTF-8
% in code blocks unless told to. Map the common non-ASCII characters that
% appear inside our lstlisting blocks.
\lstset{
  extendedchars=true,
  inputencoding=utf8,
  literate=%
    {—}{{---}}1 % em-dash
    {–}{{--}}1  % en-dash
    {’}{{\textquoteright}}1
    {‘}{{\textquoteleft}}1
    {“}{{\textquotedblleft}}1
    {”}{{\textquotedblright}}1
    {…}{{\ldots}}1
    {→}{{$\rightarrow$}}1
    {≈}{{$\approx$}}1
    {×}{{$\times$}}1
}

% Page geometry — book class, slightly wider inner margin for binding
\usepackage[a4paper, margin=2.5cm, inner=3cm]{geometry}

% Micro-typography (kerning, protrusion).  Loaded after fonts so XCharter
% pulls the right protcodes from microtype's database.
\usepackage{microtype}

% Bibliography — per-chapter refsections, biber backend
\usepackage[
  backend=biber,
  style=authoryear,
  refsection=chapter,
  maxbibnames=6,
  doi=true,
  url=true,
  urldate=long
]{biblatex}

% Hyperlinks
\usepackage[
  colorlinks=true,
  linkcolor=blue!60!black,
  citecolor=green!50!black,
  urlcolor=blue!70!black,
  pdftitle={Mastering Claude Code},
  pdfauthor={Claude Code Documentation Project}
]{hyperref}

% Cross-references with capitalization control.  American English form
% (`capitalize') — both spellings are accepted by cleveref.
\usepackage[nameinlink,capitalize]{cleveref}

% Tables — tabularx and longtable extend booktabs; booktabs itself is
% loaded by gimenezocano.sty.
\usepackage{array}
\usepackage{longtable}
\usepackage{tabularx}

% Headers and footers
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE,RO]{\thepage}
\fancyhead[LO]{\nouppercase{\rightmark}}
\fancyhead[RE]{\nouppercase{\leftmark}}
\renewcommand{\headrulewidth}{0.4pt}

% Glossary — the modern `glossaries-extra' package.  gimenezocano.sty
% intentionally does NOT load `glossaries' to avoid a package collision.
\usepackage[
  toc,
  acronym,
  section=chapter,
  nonumberlist
]{glossaries-extra}

% Index
\usepackage{imakeidx}
\makeindex[columns=2, title=Index, intoc]

% Chapter and part title styling.  gimenezocano.sty styles \section and
% deeper levels but leaves \chapter and \part to us, so the book class's
% chapter style is customised here.
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries}
  {\chaptertitlename\ \thechapter}{20pt}{\Huge}
\titleformat{\part}[display]
  {\normalfont\Huge\bfseries\centering}
  {\partname\ \thepart}{20pt}{\Huge}

% List spacing — tighten itemize and enumerate (gimenezocano styles the
% bullet markers; we keep the spacing here so it applies project-wide).
\usepackage{enumitem}
\setlist[itemize]{noitemsep, topsep=4pt}
\setlist[enumerate]{noitemsep, topsep=4pt}

% Figures
\usepackage{graphicx}
\usepackage{float}
\usepackage{caption}
\captionsetup{font=small, labelfont=bf}

% TikZ libraries — gimenezocano.sty loads tikz itself; we extend with
% the libraries used by the in-chapter diagrams.
\usetikzlibrary{
  shapes.geometric,
  shapes.misc,
  arrows.meta,
  positioning,
  fit,
  calc,
  backgrounds,
  decorations.pathreplacing,
  chains
}

% Shared TikZ styles used by the chapter diagrams. Defined here so each
% diagram can reuse them without redeclaring.
\tikzset{
  cc/.style={draw=Primary, fill=PaleBlue, rounded corners=4pt,
             minimum width=2.4cm, minimum height=0.9cm, align=center,
             font=\small\sffamily, thick},
  cchub/.style={cc, fill=Accent!50!white, minimum width=3cm, minimum height=1.2cm,
                font=\bfseries\sffamily},
  ccext/.style={cc, fill=ghmuted, draw=ghmutedrule, font=\footnotesize\sffamily},
  cclabel/.style={font=\footnotesize\sffamily\itshape, text=GrayText},
  ccflow/.style={-{Stealth[length=2.4mm]}, thick, draw=Primary},
  ccsoft/.style={-{Stealth[length=2.4mm]}, draw=ghmutedrule, dashed, thick}
}
