\section{Command-Line Cheat Sheet}
\label{app:cli-cheat}

\begin{table}[ht]
\centering
\caption{Most-used \texttt{claude} CLI invocations}
\label{tab:app-cli}
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Invocation} & \textbf{What it does} \\
\midrule
\texttt{claude}                                  & Start an interactive session in the current directory \\
\texttt{claude --model claude-opus-4-7}          & Override the session's model \\
\texttt{claude --permission-mode plan}           & Start in plan mode (read-only) \\
\texttt{claude --permission-mode auto}           & Auto mode — classifier-gated approvals \\
\texttt{claude --continue}                       & Resume the most recent session in this directory \\
\texttt{claude --resume}                         & Open the session picker \\
\texttt{claude --from-pr <N>}                    & Resume the session associated with a PR \\
\texttt{claude --worktree <name>}                & Open a session in a named git worktree \\
\texttt{claude --add-dir <path>}                 & Grant file access to an additional directory \\
\texttt{claude --plugin-dir ./<plugin>}          & Load a local plugin for this session \\
\texttt{claude --plugin-url <zip-url>}           & Load a plugin from a zip URL \\
\texttt{claude --append-system-prompt "<text>"}  & Append to the system prompt \\
\texttt{claude -p "<prompt>"}                    & Non-interactive (headless) mode \\
\texttt{claude -p "..." --output-format json}    & Headless with structured output \\
\texttt{claude mcp add --transport http <n> <u>} & Add an HTTP MCP server \\
\texttt{claude mcp add -- <command>}             & Add a stdio MCP server \\
\texttt{claude mcp list}                         & List configured MCP servers \\
\texttt{claude mcp serve}                        & Run Claude Code itself as an MCP server \\
\bottomrule
\end{tabularx}
\end{table}

\section{Troubleshooting Quick Reference}
\label{app:trouble}

\begin{table}[ht]
\centering
\caption{Common symptoms and likely fixes}
\label{tab:app-trouble}
\begin{tabularx}{\textwidth}{Xl}
\toprule
\textbf{Symptom} & \textbf{Likely cause / fix} \\
\midrule
Claude ignores a CLAUDE.md rule
  & File over 200 lines — split into rules; rule too vague; check \texttt{/memory} \\
A hook isn't firing
  & Run \texttt{/hooks} — if not listed, JSON failed to parse \\
An MCP server says ``not authenticated''
  & Run \texttt{/mcp} and complete the OAuth flow \\
\texttt{/cost} shows a session ballooning
  & Run \texttt{/context}; usually a long CLAUDE.md or many open files \\
A skill won't auto-invoke
  & \texttt{description} field too vague; \texttt{disable-model-invocation} may be \texttt{true} \\
Cross-reference shows \texttt{??}
  & Label mismatch; run \texttt{biber} then \texttt{pdflatex} twice \\
Subagent loses important state
  & Enable persistent memory: \texttt{memory: project} in the frontmatter \\
\bottomrule
\end{tabularx}
\end{table}
